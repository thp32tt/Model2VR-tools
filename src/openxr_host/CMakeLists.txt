cmake_minimum_required(VERSION 3.24)
project(Model2VROpenXRHost LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "OpenXR host requires Windows x64")
endif()
include(FetchContent)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(openxr_sdk GIT_REPOSITORY https://github.com/KhronosGroup/OpenXR-SDK.git GIT_TAG release-1.1.63 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(openxr_sdk)
add_executable(Model2OpenXRHost main.cpp)
target_compile_features(Model2OpenXRHost PRIVATE cxx_std_20)
target_compile_definitions(Model2OpenXRHost PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(Model2OpenXRHost PRIVATE openxr_loader d3d11 dxgi gdi32 user32)
if(MSVC)
  target_compile_options(Model2OpenXRHost PRIVATE /W4 /EHsc)
endif()
